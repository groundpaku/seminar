# 作業ディレクトリの作成
mkdir /home/furuya
cd /home/furuya

# Python仮想環境の作成
python3.○○ -m venv venv

# Pythonライブラリのインストール
wget https://github.com/groundpaku/seminar/blob/main/04_API/programs/fastapi/requirements.txt
source venv/bin/activate
pip install -r requirements.txt

# プログラムの配備
WinSCP等のソフトを使用して、作成したファイルを作業ディレクトリに配備してください。
・models
・routers
・schemas
・auth.py
・db.py
・main.py

# FastAPIの起動確認
uvicorn main:app --host 0.0.0.0 --port 5001
※ポート番号は重複しないように注意してください。
本社　5000番台
岡山　6000番台
倉吉　7000番台
広島　8000番台
起動確認ができたら、Ctrl + Cで停止してください。

# FastAPIのサービス設定ファイルの作成
vim /etc/systemd/system/furuya_api.service
※を参考にして作成してください。

# FastAPIのサービス起動
systemctl daemon-reload
systemctl start furuya_api.service
systemctl status furuya_api.service
※activeになっていない場合は、設定ファイル等を見直してください。
systemctl enable furuya_api.service

# Nginxの設定
vim /etc/nginx/default.d/furuya.conf
※を参考にして作成してください。

# Nginxの再起動
systemctl daemon-reload
systemctl restart nginx.service
systemctl status nginx.service
※activeになっていない場合は、設定ファイル等を見直してください。
